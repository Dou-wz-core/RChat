cmake_minimum_required(VERSION 3.16)
project(chat_udp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /utf-8 /EHsc)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
find_package(Threads REQUIRED)
add_library(chat_udp src/net/protocol.cpp src/net/reliable_udp.cpp src/net/udp_socket.cpp src/net/udp_node.cpp)
target_include_directories(chat_udp PUBLIC include)
target_link_libraries(chat_udp PUBLIC Threads::Threads)
if(WIN32)
    target_link_libraries(chat_udp PUBLIC ws2_32)
endif()
add_executable(UdpEchoServer src/tools/udp_echo_server.cpp)
target_link_libraries(UdpEchoServer PRIVATE chat_udp)
add_executable(UdpClient src/client/main.cpp)
target_include_directories(UdpClient PRIVATE thirdparty)
target_link_libraries(UdpClient PRIVATE chat_udp)
option(CHAT_BUILD_SERVER "Build MySQL chat business over UDP" ON)
if(CHAT_BUILD_SERVER)
    find_path(MYSQL_INCLUDE_DIR mysql.h HINTS "C:/Program Files/MySQL/MySQL Server 8.0/include" PATH_SUFFIXES mysql mariadb)
    find_library(MYSQL_LIBRARY NAMES libmysql mysqlclient mariadb HINTS "C:/Program Files/MySQL/MySQL Server 8.0/lib")
    if(NOT MYSQL_INCLUDE_DIR OR NOT MYSQL_LIBRARY)
        message(FATAL_ERROR "MySQL development files missing. Install libmysqlclient-dev/libmariadb-dev, or set CHAT_BUILD_SERVER=OFF for UDP tools/tests.")
    endif()
    add_subdirectory(src)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(udp_tests test/udp/udp_tests.cpp)
    target_link_libraries(udp_tests PRIVATE chat_udp)
    add_test(NAME udp_protocol_and_transport COMMAND udp_tests)
    set_tests_properties(udp_protocol_and_transport PROPERTIES TIMEOUT 30)
    if(CHAT_BUILD_SERVER)
        add_executable(chat_business_tests test/udp/chat_business_tests.cpp)
        target_link_libraries(chat_business_tests PRIVATE chat_business)
        add_test(NAME udp_chat_business_adapter COMMAND chat_business_tests)
        set_tests_properties(udp_chat_business_adapter PROPERTIES TIMEOUT 30)
    endif()
endif()
